var game
var checks := 0
var failures := 0

func snapshot() -> Dictionary:
	var m=game.model
	var visual: Vector2=m.p+(m.dance_pose() if m.perched else Vector2.ZERO)-Vector2(m.camera_x,0)
	var body:=Rect2(visual-Vector2(62,52),Vector2(124,104))
	var captions=[]
	for f in game.world._feedback:
		if f.kind!="caption" or game.world.feedback_font==null: continue
		var w: float=game.world.feedback_font.get_string_size(f.text,HORIZONTAL_ALIGNMENT_LEFT,-1,19).x
		var point:=Vector2(clampf(f.pos.x,m.camera_x+w/2+24,m.camera_x+1256-w/2)-m.camera_x,clampf(f.pos.y-45-(0.9-f.life)*24,160,670))
		captions.append(Rect2(point-Vector2(w/2+9,23),Vector2(w+18,31)))
	var cue: Rect2=game.cue_panel.get_rect()
	return {"tick":movie_ticks,"engine_frame":Engine.get_process_frames(),"stage":m.stage,"mode":game.mode,"phase":m.phase,"p":[m.p.x,m.p.y],"stamina":m.stamina,"recoveries":m.recoveries,"dew":m.garden_total(),"weather":m.weather_phase,"warning":m.warning_elapsed,"shield":m.shield_left,"shelter":m.shelter_index_at(m.p),"observed":m.observation_valid(),"small_state":m.small.state,"small_pos":[m.small.pos.x,m.small.pos.y],"keeper_state":m.keeper.state,"keeper_pos":[m.keeper.pos.x,m.keeper.pos.y],"keeper_target":[m.keeper.target.x,m.keeper.target.y],"front":m.front_open,"passage":m.passage_open,"result":m.result_elapsed,"rings":m.rings_hit,"share":m.share_state,"message_revision":m.message_revision,"listeners":m.listeners.map(func(b): return {"state":b.state,"p":[b.pos.x,b.pos.y]}),"color":m.color_step,"camera_x":m.camera_x,"viewport":[1280,800],"cue_rect":[cue.position.x,cue.position.y,cue.size.x,cue.size.y],"body_rect":[body.position.x,body.position.y,body.size.x,body.size.y],"next_shelter_screen_x":m.shelters[2].pos.x-m.camera_x,"cue_clear":game.mode!="play" or not game.cue_panel.visible or not cue.intersects(body),"cue_caption_clear":game.mode!="play" or not game.cue_panel.visible or captions.all(func(b):return not cue.intersects(b))}
func voiced(kind: String):
	event_trace.append({"tick":movie_ticks,"stage":game.model.stage,"mode":game.mode,"kind":kind})
func write_trace():
	var file=FileAccess.open(folder.path_join("trace.json"),FileAccess.WRITE)
	file.store_string(JSON.stringify({"kind":"known-solution ordinary-input route with real menu cold recovery;not independent exploration or subjective listening","fixed_fps":30,"simulation_step":1.0/60,"checks":checks,"failures":failures,"assertions":verdicts,"marks":marks,"voiced_events":event_trace,"samples":samples},"\t"))
	file.close()

func _initialize() -> void: call_deferred("run")

func check(ok: bool,label: String) -> void:
	checks += 1
	verdicts.append({"tick":movie_ticks,"pass":ok,"label":label})
	if not ok:
		failures += 1
		print("FAIL: "+label)

func key(code: Key,down: bool) -> void:
	var event := InputEventKey.new()
	event.physical_keycode = code
	event.keycode = code
	event.pressed = down
	root.push_input(event)
	if down: pressed[code] = true
	else: pressed.erase(code)

func tap(code: Key) -> void:
	key(code,true)
	key(code,false)

func release_all() -> void:
	for code in pressed.keys(): key(code,false)

func action(command: Dictionary) -> void:
	var d: Vector2 = command.direction
	var codes := [KEY_LEFT,KEY_RIGHT,KEY_UP,KEY_DOWN,KEY_SPACE]
	var values := [d.x < -0.1,d.x > 0.1,d.y < -0.1,d.y > 0.1,command.held]
	for i in codes.size():
		if values[i] != pressed.has(codes[i]): key(codes[i],values[i])

func tick(count := 1) -> void:
	for i in count:
		game.tick(1.0/60)
		game.world._process(1.0/60)
		movie_ticks+=1
		if movie_ticks%2==0:
			samples.append(snapshot())
			await process_frame

func shot(name: String) -> void:
	marks.append({"name":name,"state":snapshot()})
	if DisplayServer.get_name() == "headless": return
	game.queue_redraw()
	game.world.queue_redraw()
	await RenderingServer.frame_post_draw
	check(root.get_texture().get_image().save_png(folder.path_join(name+".png")) == OK,"capture "+name)

func new_scene() -> void:
	game = NativeFixture.new()
	game.progress_file=folder.path_join("save.json")
	root.add_child(game)
	await process_frame
	game.set_process(false)
	game.world.set_process(false)
	game.muted=false
	game.sound.set_muted(false)
	game.sound.event_played.connect(voiced)

func fly(target: Vector2) -> void:
	for frame in 1500:
		var d: Vector2 = target-game.model.p
		if d.length() < 10 or game.model.phase != "playing": break
		action({"direction":Vector2(signf(d.x) if absf(d.x)>6 else 0.0,signf(d.y) if absf(d.y)>6 else 0.0),"held":false})
		await tick()
	var reached: bool=game.model.p.distance_to(target)<10
	var exited: bool=target.x>=game.model.exit_pos.x and game.model.phase!="playing" and game.mode in ["transition","arrival","ending"]
	check(reached or exited,"waypoint or expected exit: stage%d target%s actual%s mode%s" % [game.model.stage,target,game.model.p,game.mode])
	release_all()
	await tick(24)

func cold_resume(label: String) -> void:
	tap(KEY_ESCAPE)
	var before: Dictionary = game.model.save_payload()
	check(game.mode=="pause" and game.has_saved,"actual disk pause "+label)
	await tick(90)
	check(game.model.save_payload()==before,"pause freezes transaction and listeners "+label)
	release_all()
	game.queue_free()
	await process_frame
	await new_scene()
	check(not game.saved_payload.is_empty(),"cold launch reads save "+label)
	tap(KEY_DOWN)
	tap(KEY_ENTER)
	check(game.mode=="pause" and game.model.save_payload()==before,"exact cold resume "+label)
	tap(KEY_ESCAPE)

func run() -> void:
	DirAccess.make_dir_recursive_absolute(folder)
	await new_scene()
	tap(KEY_ENTER)
	for point in [Vector2(620,520),Vector2(1250,520),Vector2(1350,400),Vector2(1540,400),Vector2(1760,520),Vector2(2170,520),Vector2(2330,520)]:await fly(point)
	await shot("01-safe-choices")
	for point in [Vector2(2330,630),Vector2(2630,600)]:await fly(point)
	await shot("02-low-rest")
	for point in [Vector2(2780,600),Vector2(2860,520),Vector2(3010,520),Vector2(3158,520),Vector2(3158,435)]:await fly(point)
	await shot("03-connected-common-roof")
	write_trace()
	print("RAIN LAYOUT PREVIEW: %d checks, %d failures"%[checks,failures])
	quit(1 if failures else 0)
