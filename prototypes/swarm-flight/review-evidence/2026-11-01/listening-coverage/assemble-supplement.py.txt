# Rebuild only the declared listening excerpts. No audio normalization.
# Usage: python3 script.py continuous.wav negative.wav recipe.json output.wav
import array, json, sys, wave
paths = dict(zip(["continuous", "negative"], sys.argv[1:3]))
recipe = json.load(open(sys.argv[3]))
sources = {}
for name, path in paths.items():
    with wave.open(path) as w:
        assert (w.getframerate(), w.getnchannels(), w.getsampwidth()) == (48000, 2, 2)
        sources[name] = array.array("h", w.readframes(w.getnframes()))
out = array.array("h")
for index, segment in enumerate(recipe["segments"]):
    if index:
        out.extend([0] * int(0.6 * 48000 * 2))
    start, end = segment["source_seconds"]
    buf = sources[segment["source"]][int(start * 48000) * 2:int(end * 48000) * 2]
    frames = len(buf) // 2
    for frame in range(480):
        gain = frame / 479
        for channel in range(2):
            buf[frame * 2 + channel] = round(buf[frame * 2 + channel] * gain)
            tail = (frames - 1 - frame) * 2 + channel
            buf[tail] = round(buf[tail] * gain)
    out.extend(buf)
with wave.open(sys.argv[4], "wb") as w:
    w.setnchannels(2)
    w.setsampwidth(2)
    w.setframerate(48000)
    w.writeframes(out.tobytes())
