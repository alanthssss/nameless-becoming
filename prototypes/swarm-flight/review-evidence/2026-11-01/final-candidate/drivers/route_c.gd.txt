func snapshot() -> Dictionary:
	var m=game.model
	return {"tick":movie_ticks,"engine_frame":Engine.get_process_frames(),"stage":m.stage,"mode":game.mode,"phase":m.phase,"p":[m.p.x,m.p.y],"velocity":[m.velocity.x,m.velocity.y],"stamina":m.stamina,"recoveries":m.recoveries,"burst_left":m.burst_left,"charge":m.charge,"charging":m.charging,"checkpoint":m.checkpoint_index,"rings":m.rings_hit,"camera_x":m.camera_x}
func check(ok: bool,label: String):
	checks+=1
	verdicts.append({"tick":movie_ticks,"pass":ok,"label":label})
	if not ok:
		failures+=1
		print("FAIL: "+label)
func voiced(kind: String): event_trace.append({"tick":movie_ticks,"stage":game.model.stage,"kind":kind})
func key(code: Key,down: bool):
	var e:=InputEventKey.new()
	e.physical_keycode=code
	e.keycode=code
	e.pressed=down
	root.push_input(e)
	if down: pressed[code]=true
	else: pressed.erase(code)
func tap(code: Key):
	key(code,true)
	key(code,false)
func release_all():
	for code in pressed.keys(): key(code,false)
func action(command: Dictionary):
	var d: Vector2=command.direction
	var codes=[KEY_LEFT,KEY_RIGHT,KEY_UP,KEY_DOWN,KEY_SPACE]
	var values=[d.x<-.1,d.x>.1,d.y<-.1,d.y>.1,command.held]
	for i in codes.size():
		if values[i]!=pressed.has(codes[i]): key(codes[i],values[i])
func tick(count:=1):
	for i in count:
		if movie_ticks>=MAX_TICKS:
			check(false,"bounded comparison simulation budget exceeded")
			write_trace()
			quit(1)
			return
		game.tick(1.0/60)
		game.world._process(1.0/60)
		movie_ticks+=1
		if movie_ticks%2==0:
			samples.append(snapshot())
			if not dry_fast: await process_frame
func observed_wait(count: int,label: String):
	var start:=movie_ticks
	await tick(count)
	waits.append({"start_tick":start,"end_tick":movie_ticks,"reason":label})
func shot(name: String):
	marks.append({"name":name,"state":snapshot()})
	if DisplayServer.get_name()=="headless": return
	game.queue_redraw()
	game.world.queue_redraw()
	await RenderingServer.frame_post_draw
	check(root.get_texture().get_image().save_png(folder.path_join(name+".png"))==OK,"capture "+name)
func new_scene():
	game=NativeFixture.new()
	game.progress_file=folder.path_join("save.json")
	root.add_child(game)
	await process_frame
	game.set_process(false)
	game.world.set_process(false)
	game.muted=false
	game.sound.set_muted(false)
	game.sound.event_played.connect(voiced)

	tap(KEY_ESCAPE)
	var before: Dictionary=game.model.save_payload()
	check(game.mode=="pause" and game.has_saved,"actual disk pause "+label)
	await observed_wait(90,"paused comparison roundtrip: "+label)
	check(game.model.save_payload()==before,"paused model exactly frozen "+label)
	release_all()
	game.queue_free()
	await process_frame
	await new_scene()
	check(not game.saved_payload.is_empty(),"disk cold scene reads actual saved payload "+label)
	tap(KEY_DOWN)
	tap(KEY_ENTER)
	if before.get("completed",false):
		check(game.mode=="ending" and game.model.save_payload()==before,"completed arrival cold-resumes at exact ending "+label)
		return
	check(game.mode=="pause" and game.model.save_payload()==before,"exact actual-menu cold resume "+label)
	tap(KEY_ESCAPE)
