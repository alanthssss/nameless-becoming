extends "res://spring_journey_verify.gd"
var records: Array=[]
func check(ok: bool,label: String) -> void:
	super.check(ok,label)
	records.append({"label":label,"pass":ok})
func prepared():
	var m=fresh()
	chapter03(m,true,false)
	chapter04(m,false)
	m.load_stage(2)
	open_front(m)
	open_back(m,false)
	for point in [Vector2(3460+O,300),Vector2(3460+O,620),Vector2(3260+O,620),Model.COMB]: fly(m,point)
	frames(m,60)
	check(m.perched and m.observation_valid() and m.passage_open,"ordinary full journey prepared valid source sharing")
	return m
func run() -> void:
	var original=prepared()
	var baseline: Dictionary=original.save_payload()
	for moment in ["armed","before_commit","after_commit","foraging"]:
		var m=fresh()
		check(m.restore_payload(JSON.parse_string(JSON.stringify(baseline,"",true,true))),"load ordinary prepared state "+moment)
		frames(m,1,Vector2.ZERO,true)
		if moment!="armed": frames(m,1)
		if moment=="before_commit": frames(m,25)
		if moment=="after_commit": frames(m,90)
		if moment=="foraging": frames(m,390)
		var committed: int=m.message_revision
		check(committed==0 if moment in ["armed","before_commit"] else committed==1,"actual precondition "+moment)
		check(m.listeners.all(func(b): return b.state=="foraging") if moment=="foraging" else true,"actual foraging precondition "+moment)
		m.invalidate_source()
		check(not m.observation_valid() and m.share_state=="idle","source close cancels armed or dancing action "+moment)
		roundtrip(m,"closed source "+moment)
		frames(m,600)
		check(m.listeners.all(func(b): return b.state=="waiting"),"source close leaves no continuing foragers "+moment)
		frames(m,1,Vector2.ZERO,true)
		frames(m,180)
		check(m.share_state=="idle" and m.message_revision==committed,"closed source cannot create a new report "+moment)
		roundtrip(m,"closed source settled "+moment)
	var leaving=fresh()
	check(leaving.restore_payload(baseline),"restore before leaving sharing")
	frames(leaving,1,Vector2.ZERO,true)
	frames(leaving,1)
	frames(leaving,25)
	frames(leaving,1,Vector2.UP)
	check(leaving.message_revision==0 and leaving.share_state=="idle" and leaving.listeners.all(func(b):return b.state=="waiting"),"ordinary movement before commit cancels without listeners")
	roundtrip(leaving,"early departure")
	var out=FileAccess.open("/private/tmp/nameless-pond-journey-rebuild/outputs/swarm-flight/2026-11-01-pond-rebuild/final/source-expiry-supplement/trace.json",FileAccess.WRITE)
	out.store_string(JSON.stringify({"revision":"75b6bda6142a36e4daa973c5f38d29e8c7f3e4b0","kind":"headless model supplement; full ordinary model steering prepares source/route, then controlled source-close content event; not blind play or natural expiry","checks":checks,"failures":failures,"assertions":records},"\t",true,true))
	out.close()
	print("SOURCE EXPIRY SUPPLEMENT VERIFIED: %d checks, %d failures" % [checks,failures])
	quit(1 if failures else 0)
